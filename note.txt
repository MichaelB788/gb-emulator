# Logging function
const struct cpu *cpu = &state->gb.cpu;
fprintf(state->log_file,
        "A:%02X F:%02X B:%02X C:%02X D:%02X E:%02X H:%02X L:%02X SP:%04X "
        "PC:%04X PCMEM:%02X,%02X,%02X,%02X\n",
        cpu->A, cpu->F, cpu->B, cpu->C, cpu->D, cpu->E, cpu->H, cpu->L,
        cpu->SP, cpu->PC, gb_read_bus(&state->gb, cpu->PC),
        gb_read_bus(&state->gb, cpu->PC + 1),
        gb_read_bus(&state->gb, cpu->PC + 2),
        gb_read_bus(&state->gb, cpu->PC + 3));
fflush(state->log_file);
